/* =============================== STAGE 1: STATE & LISTENERS ================================ */ const state = { branches: [], technicians: [], visits: [], checklists: [] }; // عناصر Dashboard const el = { branches: document.getElementById('statBranches'), technicians: document.getElementById('statTechnicians'), visits: document.getElementById('statVisits'), checklists: document.getElementById('statChecklists'), completed: document.getElementById('statCompletedVisits'), pending: document.getElementById('statPendingVisits'), visitedTechs: document.getElementById('statVisitedTechnicians'), brands: document.getElementById('statBrands') }; // Firestore listeners (بدون orderBy – آمن) function startDashboardListeners() { onSnapshot(collection(db, 'branches'), snap => { state.branches = snap.docs.map(d => ({ id: d.id, ...d.data() })); renderDashboard(); }); onSnapshot(collection(db, 'technicians'), snap => { state.technicians = snap.docs.map(d => ({ id: d.id, ...d.data() })); renderDashboard(); }); onSnapshot(collection(db, 'siteVisits'), snap => { state.visits = snap.docs.map(d => ({ id: d.id, ...d.data() })); renderDashboard(); }); onSnapshot(collection(db, 'branch_checklists'), snap => { state.checklists = snap.docs.map(d => ({ id: d.id, ...d.data() })); renderDashboard(); }); } /* =============================== STAGE 2: DASHBOARD METRICS ================================ */ function renderDashboard() { // إجماليات el.branches.textContent = state.branches.length; el.technicians.textContent = state.technicians.length; el.visits.textContent = state.visits.length; el.checklists.textContent = state.checklists.length; // زيارات مكتملة const completedVisits = state.visits.filter(v => String(v.status || '').toLowerCase() === 'completed' || v.checklistCompleted === true ); el.completed.textContent = completedVisits.length; // زيارات معلّقة const pendingVisits = state.visits.filter(v => { const s = String(v.status || '').toLowerCase(); return s !== 'completed' && s !== 'cancelled'; }); el.pending.textContent = pendingVisits.length; // فنيين قاموا بزيارة const visitedTechs = new Set( state.visits .map(v => v.technicianDocId || v.technicianId || v.technicianEmail) .filter(Boolean) ); el.visitedTechs.textContent = visitedTechs.size; // عدد البراندات من الزيارات const brands = new Set( state.visits .map(v => (v.brand || '').trim()) .filter(Boolean) ); el.brands.textContent = brands.size; }
