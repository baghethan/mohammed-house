// ===== Login Protection Config ===== const MAX_ATTEMPTS = 3; const LOCK_TIME_MS = 10 * 60 * 1000; // 10 دقائق let loginAttempts = Number(localStorage.getItem('loginAttempts') || 0); let lockUntil = Number(localStorage.getItem('lockUntil') || 0); // ===== Helper ===== function isLocked() { return Date.now() < lockUntil; } function lockLogin() { lockUntil = Date.now() + LOCK_TIME_MS; localStorage.setItem('lockUntil', lockUntil); $('loginBtn').disabled = true; $('loginMsg').textContent = '🚫 تم قفل تسجيل الدخول بعد 3 محاولات. انتظر 10 دقائق ثم حاول مرة واحدة فقط.'; } function resetLoginAttempts() { loginAttempts = 0; lockUntil = 0; localStorage.removeItem('loginAttempts'); localStorage.removeItem('lockUntil'); $('loginBtn').disabled = false; } // ===== On Page Load: check lock ===== if (isLocked()) { $('loginBtn').disabled = true; const remaining = Math.ceil((lockUntil - Date.now()) / 60000); $('loginMsg').textContent = `⏳ تسجيل الدخول مقفل. حاول بعد ${remaining} دقيقة`; } // ===== Login Handler ===== $('loginBtn').onclick = async () => { if (isLocked()) { const remaining = Math.ceil((lockUntil - Date.now()) / 60000); $('loginMsg').textContent = `⏳ تسجيل الدخول مقفل. حاول بعد ${remaining} دقيقة`; return; } const email = $('loginEmail').value.trim(); const password = $('loginPassword').value; if (!email || !password) { $('loginMsg').textContent = '⚠️ أدخل البريد الإلكتروني وكلمة المرور'; return; } try { $('loginBtn').disabled = true; $('loginMsg').textContent = '🔐 جاري التحقق...'; await signInWithEmailAndPassword(auth, email, password); // ✅ نجاح: إعادة تعيين العداد resetLoginAttempts(); } catch (e) { console.error(e); loginAttempts++; localStorage.setItem('loginAttempts', loginAttempts); if (loginAttempts >= MAX_ATTEMPTS) { lockLogin(); return; } // ✅ رسائل واضحة if (e.code === 'auth/wrong-password') { $('loginMsg').textContent = `❌ كلمة المرور غير صحيحة (${loginAttempts}/3)`; } else if (e.code === 'auth/user-not-found') { $('loginMsg').textContent = `❌ المستخدم غير موجود (${loginAttempts}/3)`; } else if (e.code === 'auth/quota-exceeded') { $('loginMsg').textContent = '🚫 تم تجاوز عدد المحاولات من Firebase. انتظر قليلًا.'; } else { $('loginMsg').textContent = `❌ فشل تسجيل الدخول (${loginAttempts}/3)`; } $('loginBtn').disabled = false; } };
