// --- وظيفة استيراد الموظفين المطابقة لملفك تماماً --- document.getElementById('importEmpFile').onchange = (e) => { const file = e.target.files[0]; const reader = new FileReader(); reader.onload = async (ev) => { try { const data = ev.target.result; const workbook = XLSX.read(data, { type: 'binary' }); const sheet = workbook.Sheets[workbook.SheetNames[0]]; const rows = XLSX.utils.sheet_to_json(sheet); if (rows.length === 0) return alert("الملف فارغ!"); let count = 0; for (let row of rows) { const keys = Object.keys(row); const findV = (names) => { const k = keys.find(k => names.includes(k.toLowerCase().trim())); return k ? row[k] : null; }; // تحديث الاحتمالات لتشمل empid التي استخدمتها في ملفك const id = findV(["empid", "id", "emp id", "employee id", "رقم الموظف"]); const name = findV(["name", "full name", "الاسم", "employee name"]); const dept = findV(["dept", "department", "القسم"]); const role = findV(["role", "job title", "المسمى"]); if (id && name) { await addDoc(empCol, { empId: String(id).trim(), name: String(name).trim(), dept: String(dept || "").trim(), role: String(role || "Employee").trim() }); count++; } } if (count > 0) { alert(`تم استيراد ${count} موظف بنجاح! ✅`); } else { alert("لم يتم العثور على بيانات! تأكد أن السطر الأول في الاكسيل يحتوي على empId و name."); } } catch (err) { alert("خطأ في قراءة ملف الاكسيل."); } }; reader.readAsBinaryString(file); };
