// --- وظيفة استيراد الموظفين الاحترافية (حل مشكلة الموظف الواحد) --- document.getElementById('importEmpFile').onchange = (e) => { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = async (ev) => { try { const data = ev.target.result; const workbook = XLSX.read(data, { type: 'binary' }); const sheet = workbook.Sheets[workbook.SheetNames[0]]; // تحويل البيانات مع التأكد من إزالة أي مسافات زائدة في العناوين const rows = XLSX.utils.sheet_to_json(sheet, { defval: "" }); let successCount = 0; let errorCount = 0; console.log("Total rows found in Excel:", rows.length); for (let i = 0; i < rows.length; i++) { const row = rows[i]; const keys = Object.keys(row); // وظيفة ذكية للبحث عن العمود الصحيح مهما كان مسمى حرفه (كبير أو صغير) const findVal = (list) => { const k = keys.find(key => list.includes(key.toLowerCase().trim())); return k ? String(row[k]).trim() : null; }; const id = findVal(["empid", "id", "employee id", "رقم الموظف"]); const name = findVal(["name", "full name", "الاسم", "الاسم الكامل"]); const dept = findVal(["dept", "department", "القسم"]) || ""; const role = findVal(["role", "job title", "المسمى"]) || "Employee"; // التحقق من وجود البيانات الأساسية قبل الإرسال لـ Firebase if (id && name && id !== "" && name !== "") { try { // استخدام await هنا "إجباري" لضمان عدم تخطي الموظفين await addDoc(empCol, { empId: id, name: name, dept: dept, role: role, importedAt: new Date() // علامة زمنية للتدقيق }); successCount++; console.log(`Successfully added: ${name}`); } catch (fireErr) { console.error(`Firebase error at row ${i+2}:`, fireErr); errorCount++; } } else { console.warn(`Row ${i+2} skipped: Missing ID or Name`, row); } } alert(`Import Finished! 🚀\n✅ Success: ${successCount}\n❌ Failed/Skipped: ${errorCount}`); } catch (err) { console.error("General Import Error:", err); alert("Error reading Excel file. Make sure it's a valid .xlsx file."); } }; reader.readAsBinaryString(file); };
