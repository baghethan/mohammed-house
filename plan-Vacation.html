// --- وظيفة استيراد الموظفين المطورة (Flexible Import) --- document.getElementById('importEmpFile').onchange = (e) => { const file = e.target.files[0]; const reader = new FileReader(); reader.onload = async (ev) => { try { const data = ev.target.result; const workbook = XLSX.read(data, { type: 'binary' }); const sheet = workbook.Sheets[workbook.SheetNames[0]]; const rows = XLSX.utils.sheet_to_json(sheet); if (rows.length === 0) { alert("The file appears to be empty!"); return; } let count = 0; for (let row of rows) { // تحويل مفاتيح الصف لنصوص صغيرة وبدون مسافات للبحث بذكاء const keys = Object.keys(row); const findValue = (possibleNames) => { const foundKey = keys.find(k => possibleNames.includes(k.toLowerCase().trim())); return foundKey ? row[foundKey] : null; }; // البحث عن القيم بمرونة عالية const id = findValue(["id", "employee id", "رقم الموظف", "الرقم", "emp id"]); const name = findValue(["name", "full name", "الاسم", "الاسم الكامل", "employee name"]); const dept = findValue(["dept", "department", "القسم", "الإدارة"]); const role = findValue(["role", "job title", "المسمى", "المسمى الوظيفي"]) || "Employee"; if (id && name) { await addDoc(empCol, { empId: String(id).trim(), name: String(name).trim(), dept: String(dept || "").trim(), role: String(role).trim() }); count++; } } if (count > 0) { alert(`Successfully imported ${count} employees! ✅`); } else { alert("Imported 0 employees. Please check if your Excel columns are named 'ID' and 'Name'."); console.log("Sample row from your file:", rows[0]); // يطبع في Console لتفحص المشكلة } } catch (err) { console.error("Import Error:", err); alert("Error reading the Excel file."); } }; reader.readAsBinaryString(file); };
